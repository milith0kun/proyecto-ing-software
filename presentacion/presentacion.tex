\begin{frame}{1. Problemática y Justificación del Proyecto}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{academicbox}{Situación Previa en CGB Academy}
            \scriptsize
            \begin{itemize}
                \item \textbf{Dispersión de Procesos:} Tres marcas formativas (CIIP, GEOMINA, BIOMEDIC) con metodologías de inducción no estandarizadas.
                \item \textbf{Costos en Horas Hombre:} Inducción manual mediante archivos PDF estáticos y reuniones verbales repetitivas.
                \item \textbf{Falta de Trazabilidad:} Ausencia de métricas sobre la comprensión y el progreso del personal ingresante.
                \item \textbf{Inviabilidad de LMS Pesados:} Sistemas tradicionales tipo Moodle agregan complejidad técnica y sobrecarga de servidor.
            \end{itemize}
        \end{academicbox}
    \end{column}
    \begin{column}{0.48\textwidth}
        \begin{formalbox}{Solución de Ingeniería Propuesta}
            \scriptsize
            \begin{itemize}
                \item \textbf{Motor de Diapositivas Ágil:} Constructor de capacitaciones interactivas paso a paso (\textit{no-code}).
                \item \textbf{Doble Ámbito Estructurado:}
                \begin{itemize}
                    \item \textit{Público:} Orientación libre para estudiantes y docentes.
                    \item \textit{Interno:} Rutas jerárquicas de onboarding por Área y Puesto.
                \end{itemize}
                \item \textbf{Microtests Formativos:} Retroalimentación pedagógica instantánea sin penalizaciones punitivas.
                \item \textbf{Infraestructura Eficiente:} Despliegue optimizado en servidor VPS (2 vCPU / 8 GB RAM).
            \end{itemize}
        \end{formalbox}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{7. Flujo Kanban y Justificación de Límites WIP}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{academicbox}{Fundamentación Matemática: Ley de Little}
            \scriptsize
            $$\text{WIP} = \text{Throughput (TH)} \times \text{Cycle Time (CT)}$$
            \begin{itemize}
                \item \textbf{Desarrollo [WIP = 3]:} 1 desarrollador activo = Máximo 1 Historia de Usuario activa (Alex, Domingo, Ronaldo).
                \item \textbf{Revisión [WIP = 2]:} Control de calidad óptimo para pruebas e integración por pares.
                \item \textbf{Principio Lean:} Cuando Revisión alcanza 2/2, se suspende el inicio de nuevo código y se prioriza terminar integraciones (\textit{``Stop starting, start finishing''}).
            \end{itemize}
        \end{academicbox}
    \end{column}
    \begin{column}{0.48\textwidth}
        \begin{formalbox}{Gestión de Bloqueos ($\oslash$ BLOQUEADA)}
            \scriptsize
            \begin{itemize}
                \item Las tareas impedidas permanecen en su columna física para evitar distorsiones en las métricas de Cycle Time.
                \item Se asigna la etiqueta $\mathbf{\oslash\text{ BLOQUEADA}}$ registrando motivo técnico, responsable y plan de acción.
                \item El \textbf{Scrum Master} asume la facilitación prioritaria para remover el impedimento durante la Daily Scrum.
            \end{itemize}
        \end{formalbox}
    \end{column}
\end{columns}
\end{frame}
