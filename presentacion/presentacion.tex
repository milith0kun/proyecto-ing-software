\begin{frame}{1. Problemática y Justificación del Proyecto}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{alertbox}{Situación Previa en CGB Academy}
            \scriptsize
            \begin{itemize}
                \item \textbf{Dispersión Institucional:} Tres marcas corporativas (CIIP, GEOMINA y BIOMEDIC) con procesos formativos desarticulados.
                \item \textbf{Proceso de Inducción Ineficiente:} Inducción manual basada en documentos PDF estáticos y charlas verbales repetitivas.
                \item \textbf{Cero Trazabilidad:} Sin registro del nivel de comprensión ni del avance real del colaborador ingresante.
                \item \textbf{Sobrecarga de LMS Clásicos:} Plataformas pesadas tipo Moodle generan fricción técnica innecesaria.
            \end{itemize}
        \end{alertbox}
    \end{column}
    \begin{column}{0.48\textwidth}
        \begin{greenbox}{Solución Propuesta: Plataforma Modular}
            \scriptsize
            \begin{itemize}
                \item \textbf{Motor de Slides Liviano:} Creación visual (\textit{no-code}) de capacitaciones interactivas paso a paso.
                \item \textbf{Doble Ámbito Operativo:}
                \begin{itemize}
                    \item \textit{Público:} Orientación libre para estudiantes y docentes.
                    \item \textit{Interno:} Onboarding jerárquico por Área y Puesto.
                \end{itemize}
                \item \textbf{Microtests Formativos:} Evaluación interactiva inmediata sin fines punitivos.
                \item \textbf{Infraestructura Eficiente:} Operación fluida en VPS (2 vCPU / 8 GB RAM).
            \end{itemize}
        \end{greenbox}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{7. Flujo Kanban y Fundamentación de Límites WIP}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{techbox}{Límites WIP y Ley de Little}
            \scriptsize
            $$\text{WIP} = \text{Throughput (TH)} \times \text{Cycle Time (CT)}$$
            \begin{itemize}
                \item \textbf{Desarrollo [WIP = 3]:} 1 desarrollador activo = Máximo 1 Historia de Usuario simultánea (Alex, Domingo, Ronaldo).
                \item \textbf{Revisión [WIP = 2]:} Amortiguador óptimo para control de calidad y pruebas de integración.
                \item \textbf{Regla Lean:} Si Revisión se satura (2/2), se detiene el inicio de nuevo código y se prioriza terminar revisiones (\textit{``Stop starting, start finishing''}).
            \end{itemize}
        \end{techbox}
    \end{column}
    \begin{column}{0.48\textwidth}
        \begin{alertbox}{Política de Bloqueos ($\oslash$ BLOQUEADA)}
            \scriptsize
            \begin{itemize}
                \item Las tareas impedidas permanecen en su columna física para no distorsionar métricas de tiempo de ciclo.
                \item Se etiquetan como $\mathbf{\oslash\text{ BLOQUEADA}}$ registrando motivo, responsable y acción mitigadora.
                \item El \textbf{Scrum Master} asume la remoción prioritaria del bloqueo durante la Daily Scrum.
            \end{itemize}
        \end{alertbox}
    \end{column}
\end{columns}
\end{frame}
